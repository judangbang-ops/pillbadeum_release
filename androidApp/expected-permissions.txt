# 디버그·릴리즈 APK의 권한 목록 기준 (aapt2 dump permissions 형식, package 줄 제외).
# CI(.github/workflows/ci.yml의 디버그·릴리즈 APK, release.yml의 릴리즈 APK)가 실제 APK와 비교해, 한 줄이라도 늘거나 줄면 실패함.
# 의도한 변경이면 이 파일을 같은 PR에서 고치고, 그 권한을 어디서 넣는지 주석으로 적음.
# android.permission.INTERNET은 이 파일과 상관없이 항상 실패로 봄 (REQ-057).
# 패키지 이름은 ${applicationId}로 적음. 패키지 이름(Q9, TASK-001)이 바뀌어도 그대로 쓰려는 것임.
# 출처: 설계 독립 검토 B8 (브랜치 claude/project-thread-j7y084 fca62ce,
#       docs/06-review/evidence-2026-11-11/build-and-apk-check.txt, 7b8d9df 빌드 기준).
# 릴리즈 APK도 같은 기준과 비교함 (TASK-002, 2026-11-12). 두 변형의 권한이 달라지면 이 파일을 변형별로 나눔.

# 앱 매니페스트: 재부팅 뒤 하루 경계 갱신과 복용 알림 알람을 다시 걺
uses-permission: name='android.permission.RECEIVE_BOOT_COMPLETED'

# 앱 매니페스트: 복용 알림 (TASK-025). Android 13 이상은 알림을 켤 때 물음 (TASK-023, SPEC-028)
uses-permission: name='android.permission.POST_NOTIFICATIONS'

# 앱 매니페스트: 허용된 동안 정확 알람으로 복용 알림을 울림, 아니면 비정확 (TASK-024, REQ-050). USE_EXACT_ALARM은 넣지 않음 (REQ-052)
uses-permission: name='android.permission.SCHEDULE_EXACT_ALARM'

# androidx.work:work-runtime 2.7.1 (Glance가 끌어옴). ACCESS_NETWORK_STATE를 뺄지는 위젯 시험 뒤 정함 (검토 B8)
uses-permission: name='android.permission.WAKE_LOCK'
uses-permission: name='android.permission.ACCESS_NETWORK_STATE'
uses-permission: name='android.permission.FOREGROUND_SERVICE'

# androidx.core 1.18.0: 앱 안에서만 쓰는 동적 수신기 권한
permission: ${applicationId}.DYNAMIC_RECEIVER_NOT_EXPORTED_PERMISSION
uses-permission: name='${applicationId}.DYNAMIC_RECEIVER_NOT_EXPORTED_PERMISSION'
